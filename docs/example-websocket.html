WebSocket Speaker ID Server

Build a real-time speaker identification API. Accept audio over WebSocket, return speaker identity.

The Code

package main

import (
    "encoding/binary"
    "fmt"
    "net/http"
    "github.com/Aswanidev-vs/go-cognitive-voice/pkg/gcv"
)

func main() {
    // Create engine
    cfg := gcv.DefaultConfig()
    cfg.MatchThreshold = 0.7
    engine := gcv.NewEngine(cfg)

    // Load existing speakers
    engine.LoadDatabase("./speaker_db")

    // Enroll speakers via POST /enroll
    http.HandleFunc("/enroll", func(w http.ResponseWriter, r *http.Request) {
        // Parse multipart form: id, name, audio files
        r.ParseMultipartForm(32 << 20)
        id := r.FormValue("id")
        name := r.FormValue("name")

        // Save uploaded files temporarily, enroll, then delete
        // ... (file handling omitted for brevity)

        fmt.Fprintf(w, "Enrolled %s\n", name)
    })

    // Identify via POST /identify (multipart: audio file)
    http.HandleFunc("/identify", func(w http.ResponseWriter, r *http.Request) {
        // Read uploaded audio file
        file, _, _ := r.FormFile("audio")
        defer file.Close()

        // Decode to samples (simplified)
        samples := decodeAudio(file)

        // Identify
        result, _ := engine.IdentifyFromSamples(samples, 16000)

        // Return JSON
        fmt..Fprintf(w, `{"speaker":"%s","score":%.4f,"identified":%v}`,
            result.SpeakerName, result.Score, result.Identified)
    })

    // Streaming: POST /stream (chunked transfer encoding)
    http.HandleFunc("/stream", func(w http.ResponseWriter, r *http.Request) {
        buf := engine.NewStreamBuffer(16000, 1.0)

        // Read chunks from request body
        chunk := make([]byte, 8000)
        for {
            n, err := r.Body.Read(chunk)
            if n > 0 {
                // Convert bytes to float64 samples
                samples := bytesToFloat64(chunk[:n])
                buf.Write(samples)

                if result, ok := buf.Identify(); ok {
                    fmt.Fprintf(w, "%s (%.1f%%)\n",
                        result.SpeakerName, result.Score*100)
                }
            }
            if err != nil { break }
        }
    })

    fmt.Println("Server on :8080")
    http.ListenAndServe(":8080", nil)
}

API Endpoints

POST /enrollRegister a speaker. Multipart form: id, name, audio files.
POST /identifyIdentify speaker from audio. Returns JSON with speaker, score, identified.
POST /streamStreaming identification. Send audio chunks, get results as they arrive.

Client Usage

# Enroll a speaker
curl -X POST http://localhost:8080/enroll \
  -F "id=alice" \
  -F "name=Alice" \
  -F "audio=@alice.wav"

# Identify
curl -X POST http://localhost:8080/identify \
  -F "audio=@mystery.wav"
# → {"speaker":"Alice","score":0.8806,"identified":true}

# Stream (send raw PCM chunks)
cat audio.pcm | curl -X POST http://localhost:8080/stream \
  --data-binary @-
# → Alice (88.1%)

Deployment

Production tips: Use gorilla/websocket for proper WebSocket support. Add authentication. Use a reverse proxy (nginx) for TLS. Store the speaker database on a persistent volume.
